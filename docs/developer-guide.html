4.1 สองภาษาด้วย data-th / data-en

ข้อความใน HTML ใส่ทั้ง 2 ภาษาเป็น attribute ข้อความตั้งต้นเป็นภาษาไทย ข้อความที่ JS สร้างใช้ t({ th, en }, params) ซึ่งแทน {name} ด้วยค่าที่ส่งมา

export function t(message, params = {}) {
  const text = message[current] ?? message.en;
  return text.replace(/\{(\w+)\}/g, (_, k) => (params[k] ?? ""));
}
// ใช้: t({ th: "{n} วันล่าสุด", en: "Last {n} days" }, { n: 7 })

05รูปแบบไฟล์ข้อมูล

ไฟล์ .log .txt .csv .xlsx มีข้อมูลเหมือนกัน ต่างกันแค่การห่อ: metadata 1 บรรทัด + header + แถวข้อมูล

5.1 ความแตกต่างของแต่ละนามสกุล

รูปแบบรายละเอียด
.log / .txtคั่นด้วย TAB ขึ้นบรรทัด CRLF
.csvตัวคั่น , ; หรือ TAB (ตรวจจาก header เอง) รองรับ "..." และ BOM ของ Excel
.xlsxใช้ sheet แรก อ่านด้วยโค้ดเขียนเอง (zip + DecompressionStream) เซลล์ว่างท้ายแถวไม่ถูกเก็บ จึงเช็กจำนวนคอลัมน์ถึงคอลัมน์สุดท้ายที่ใช้เท่านั้น · บางไฟล์ (ConvertCSV) ไม่มี metadata
CSV ดิบ ESP32แถวแรก timestamp,raw_hex แถวละ frame 228 byte เป็น hex ทุก 10 นาที (บท 10)
.xlsไม่รองรับ แจ้งให้ Save As เป็น .xlsx / .csv

5.2 คอลัมน์เวลาและรูปแบบวันที่

คอลัมน์เวลาชื่อมีช่องว่างหลายตัว (Date / Time ) จึงเทียบแบบตัดช่องว่างและไม่สนตัวพิมพ์ ชื่อที่รองรับอยู่ใน TIME_COLUMN_NAMES เจอชื่อใหม่ให้เพิ่มที่นี่ที่เดียว

ตัวอย่างรูปแบบที่มา
7/24/2026 2:53:03 PMM/D/YYYY h:mm:ss AM/PMlog 1 วินาที
16/09/2026 00:00:29DD/MM/YYYY HH:mm:ssDataLogger WLAN
2026-09-16 00:00:29ISOCSV ที่ export ใหม่ / ESP32
46281.0003เลขวันที่ของ Excel.xlsx

ห้ามใช้ new Date(string)แต่ละ browser ตีความรูปแบบ US ไม่เหมือนกัน parse วันที่ด้วย regex เอง (parseDateParts()) วัน/เดือนสลับกันได้ จึงตัดสินจากข้อมูลทั้งไฟล์: ตัวเลขแรก > 12 = D/M, ตัวที่สอง > 12 = M/D, ตัดสินไม่ได้ → มี AM/PM = M/D, 24 ชม. = D/M

07อ่านไฟล์แบบ stream (sources.js)

ไฟล์ 100 MB อ่านทั้งก้อนด้วย file.text() แล้ว split จะกิน memory หลายเท่าของขนาดไฟล์ จึงอ่านทีละส่วนแล้วตัดทีละบรรทัด

js/sources.js · readTextRecords
export async function readTextRecords(file, onRecord, onProgress) {
  const reader = file.stream().getReader();
  const decoder = new TextDecoder("utf-8");
  let leftover = "";      // เศษบรรทัดที่ถูกตัดกลางระหว่าง chunk
  let lineNo = 0, delimiter = null;

  const handleLine = (raw) => {
    lineNo++;
    let line = raw.endsWith("\r") ? raw.slice(0, -1) : raw;     // CRLF
    if (lineNo === 1 && line.charCodeAt(0) === 0xfeff) line = line.slice(1); // BOM จาก Excel
    if (!delimiter) delimiter = detectDelimiter(line);           // TAB / , / ; ที่พบมากที่สุด
    const fields = !delimiter ? [line]
      : delimiter === "\t" ? line.split("\t")
      : splitCsvLine(line, delimiter);                           // รองรับ "a,b" และ ""
    onRecord(fields, lineNo);
  };
  // วน reader.read() → decoder.decode(chunk, {stream:true}) → ตัดด้วย "\n" → handleLine
}

09ช่วงข้อมูลขาด และไฟดับ

กราฟต้องไม่ลากเส้นเชื่อมค่าจริงข้ามช่วงที่ไม่มีข้อมูล และต้องบอกได้ว่าช่วงนั้นไฟดับหรือแค่ข้อมูลไม่มา

9.1 Sampling อัตโนมัติ

ไม่มีเกณฑ์ตายตัว เพราะ log ปกติบันทึก ~1 วินาที, DataLogger WLAN ~30–90 วินาที, ESP32 ทุก 10 นาที เว็บหา sampling เอง = median ของระยะห่างระหว่างแถว (ไม่นับเวลาซ้ำ) และช่วงที่ห่างเกิน GAP_MEDIAN_FACTOR (3) × sampling = ช่วงขาด

9.2 ไฟดับ หรือข้อมูลหาย: ดูจากตัวนับ ProTestTimer

ProTestTimer นับขึ้นเป็นนาทีตั้งแต่เปิดเครื่อง ถ้าข้ามช่วงขาดแล้วตัวนับเพิ่มเท่ากับเวลาที่ผ่านไป = ตู้ทำงานตลอด แค่ข้อมูลไม่มา ถ้าตัวนับ reset หรือเพิ่มน้อยกว่า 90% = ไฟดับ

  • worker แทรก 3 แถวต่อช่วง (ต้น / กลาง / ท้าย) และจดชนิดไว้ใน gapRows (1/2 = ไฟดับ, 3/4 = ข้อมูลหาย) จากนั้น fillGaps() ใน main.js เติมค่าที่วาด
  • ทั้ง 2 แบบ: เซนเซอร์เส้นขาด · Control panel / Error ไม่วาด · Min/Max/เฉลี่ยไม่นับช่วงนี้ (ค่าดิบยังเป็น NaN)
  • ตัดสินไม่ได้ (ไม่มีตัวนับ / ตัวนับไม่เดิน / ช่วงสั้นจนตัวนับควรเพิ่มไม่ถึง 2) = ถือเป็นข้อมูลหาย
  • แถบพื้นหลังจางมาก (GAP_BAND_ALPHA 5%) เต็มความสูงช่วง และข้อความ |<-- ไฟดับ -->| วาดบน canvas จึงติดไปกับรูปที่บันทึก

9.3 Reset ของตัวนับ (counterResets())

Reset = ค่าน้อยกว่าแถวก่อน (ไม่ใช่ดูค่า 0 เพราะ sampling 10 นาทีอาจไม่เห็น 0) ยกเว้นการนับจนเต็ม (65535 → 0) ซึ่งเป็นการนับเกิน ตัดสินจากแถวก่อนอยู่ใกล้ค่าเต็มพอจะนับเกินได้ในเวลานั้น บางไฟล์นับถอยหลัง จึงตัดสินทิศการนับจากทั้งไฟล์

บั๊กที่เคยเกิด (แก้ 2026-11-04)ความเร็วการนับห้ามใช้ median ของ "ค่าที่เพิ่ม ÷ เวลาระหว่างแถว" เพราะ log 1 วินาทีบันทึกทุกวินาทีแต่ตัวนับเดินทีละนาที จะได้ค่าที่เร็วเกินจริง 60 เท่า แล้วช่วงข้อมูลหายถูกตัดสินเป็นไฟดับ ต้องใช้ ค่าที่เพิ่มทั้งหมด ÷ เวลาทั้งหมด (counterProfile())

ค่า 0 ซ้ำหลัง reset

ProTestTimer นับเป็นนาที ถ้า sampling 1 วินาที ค่า 0 จะซ้ำราว 60 แถวก่อนเป็น 1 ซึ่งเป็นการนับปกติ จึงนับ reset เฉพาะแถวที่ค่าตกลงมา (เช่น 89 → 0)

ถ้าค่า 0 ค้างนานเกิน 1 รอบนับ (POWER_COUNTER.zeroHold 1.05 × เวลาต่อ 1 ค่า) แปลว่าไฟดับซ้ำตอนต้นช่วง:

  • ตัวอย่าง: 0 × 70 → 1 คือเปิดมา 10 วินาทีแล้วดับอีก แถว 1–10 จึงเป็น reset และ 60 แถวหลังเป็น ON
  • ตัดสินได้ตอนเห็นค่า 1 เท่านั้น (closeZeroRun() ย้อนกลับไปติดป้าย reset ให้แถวต้นช่วง)
  • ช่วงค่า 0 ที่ยังไม่เป็น 1 จนจบไฟล์ ตัดสินไม่ได้ จึงตกเฉพาะแถวที่ค่าตกลงมา

function once(key, fn) {
  return withTimeout(shared(key, fn), SHARED_WAIT_MS);

function shared(key, fn) {
  const current = inflight.get(key);
  if (current && Date.now() - current.started < SHARED_WAIT_MS) return current.promise;
  const entry = { started: Date.now() };
  entry.promise = fn().finally(() => { if (inflight.get(key) === entry) inflight.delete(key); });
  inflight.set(key, entry);
  return entry.promise;
}

  if (isInactive(device, at)) return "inactive";
  if (!isOnline(device, at)) return "offline";
  if (lastRecord == null || Number.isNaN(lastRecord)) return "online";
  const recordReal = lastRecord - zoneOffset(FILE_TIME_ZONE, lastRecord); // เวลาไทย → เวลาจริง
  const modifiedReal = Date.parse(device.lastModified) / 1000;
  return modifiedReal - recordReal > OFFLINE_MINUTES * 60 ? "nodata" : "online";
}
